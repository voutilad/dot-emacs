* Clojure Support

** clojure-mode
   Major modes for Clojure, ClojureScript, and EDN.

   #+BEGIN_SRC emacs-lisp
     (use-package clojure-mode
       :ensure t
       :hook (clojure-mode . prettify-symbols-mode)
       :config
       (setq clojure-mode-pretty-symbols-alist
             '(("fn" . ?λ))))
   #+END_SRC

** Structural Editing
   Strict Smartparens with Evil integration for Clojure.
   Ordinary Smartparens for Java.

   #+BEGIN_SRC emacs-lisp
     (use-package smartparens
       :ensure t
       :demand t
       :hook (java-mode . smartparens-mode)
       :config
       (require 'smartparens-config)
       (sp-use-smartparens-bindings))

     (use-package evil-smartparens
       :ensure t
       :after (evil smartparens)
       :demand t
       :preface
       (defun my/enable-structural-editing ()
         "Enable strict Smartparens before Evil integration."
         (smartparens-strict-mode 1)
         (evil-smartparens-mode 1))
       :hook ((clojure-mode . my/enable-structural-editing)
              (clojurescript-mode . my/enable-structural-editing)
              (clojurec-mode . my/enable-structural-editing)
              (edn-mode . my/enable-structural-editing)
              (cider-repl-mode . my/enable-structural-editing)))
   #+END_SRC

* CIDER REPL
  Interactive Clojure development using nREPL.

  #+BEGIN_SRC emacs-lisp
    (use-package cider
      :ensure t
      :commands (cider-jack-in
                 cider-connect-clj
                 cider-connect-cljs)
      :init
      (setq cider-repl-history-size 1000
            cider-repl-wrap-history t
            cider-show-error-buffer nil
            cider-auto-select-error-buffer nil

            ;; Prefer IPv4 loopback on OpenBSD.
            cider-host "127.0.0.1"))
  #+END_SRC
